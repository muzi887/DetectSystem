import axios from 'axios';

const request = axios.create({

/**
 * 发送图片和数据到后端进行分析

export const analyzeImage = (data: {

  category: string, 
  additionalInfo?: string

## 前端 Vue 如何找到后端 Flask？

你提供的 `analysis.ts` 代码写得很规范，但有一个关键点：**Vue (Vite) 默认不知道后端在 5000 端口**。如果配置不对，前端请求会发到 5173 端口然后报错 404。

请按照以下两步配置，打通前后端：

### 第一步：配置 Vite 代理 (`vite.config.ts`)

你需要告诉 Vite：“凡是看到 `/api` 开头的请求，都帮我转发给 Python 后端（端口 5000）”。

打开项目根目录下的 `vite.config.ts`，找到 `server` 配置项（如果没有就加上），修改如下：

```typescript
// vite.config.ts
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
// ... 其他引入

export default defineConfig({
  plugins: [vue()],
  server: {
    // 默认端口通常是 5173
    port: 5173, 
    // 【核心配置】代理设置
    proxy: {
      '/api': {
        target: 'http://127.0.0.1:5000', // 后端地址
        changeOrigin: true, // 允许跨域
        // 关键点：不要重写路径！
        // 因为你的 Python 代码里写的是 @app.route('/api/analysis/image')
        // 所以我们要原封不动地把 /api 发过去
      }
    }
  }
})
```

### 第二步：检查 HTTP 工具封装 (`src/utils/http.ts`)

你引用了 `import http from '@/utils/http'`。请确保你的 `http.ts`（axios 封装）里的 `baseURL` 设置是正确的。

通常有两种设置方式，配合上面的代理，推荐 **方式 A**：

**方式 A（推荐）：相对路径**  
让代理生效，baseURL 应该设置为 `/api` 或者干脆留空（如果在请求里写了 `/api`）。

```typescript
// src/utils/http.ts
import axios from 'axios'

const http = axios.create({
  // 这里的 baseURL 设为 '/api'
  // 这样当你调用 http.post('/analysis/image') 时
  // 实际请求是：http://localhost:5173/api/analysis/image
  // 然后 Vite 代理把它转发给：http://127.0.0.1:5000/api/analysis/image
  baseURL: '/api', 
  timeout: 10000,
})
// ... 拦截器代码 ...
export default http
```

**如果你的 `http.ts` 是这么写的，那你现在的 `analysis.ts` 里的请求地址要微调：**

- 如果 `baseURL: '/api'`，那么 `analysis.ts` 里写 `http.post('/analysis/image', ...)` —— **正确**。
- 如果 `baseURL: ''` (空)，那么 `analysis.ts` 里写 `http.post('/api/analysis/image', ...)` —— **正确**。

## 两个后端：分流代理

我们需要告诉 Vite：**“如果是分析图片的请求，找 Flask；如果是其他数据请求，找 json-server。”**

假设你的 `json-server` 运行在 **3000** 端口（这是它的默认端口）。

修改 `vite.config.ts`，使用 **“精确匹配优先”** 的策略：

```typescript
// vite.config.ts
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'

export default defineConfig({
  plugins: [vue()],
  server: {
    port: 5173,
    proxy: {
      // 1. 先匹配最长、最具体的路径：智能分析接口 -> 转发给 Flask
      '/api/analysis': {
        target: 'http://127.0.0.1:5000', 
        changeOrigin: true,
        // 只要你的 Flask 路由是 @app.route('/api/analysis/image')，这里就不需要 rewrite
      },

      // 2. 再匹配剩下的通用路径：其他数据接口 -> 转发给 json-server
      '/api': {
        target: 'http://localhost:3000', // 假设你的 json-server 在 3000
        changeOrigin: true,
        // 【重要】json-server 通常没有 /api 前缀，需要把它去掉
        rewrite: (path) => path.replace(/^\/api/, '') 
      }
    }
  }
})
```

我们可以把这个过程想象成去**医院看病**：

- **Python 后端 (5000端口)** 是 **“专科医生”**。他只负责看片子（图片），告诉你得了什么病，但他不负责写病历，看完就忘了。

## 数据没有写入 `db.json` 
你的代码逻辑在 `DataAnalysis.vue` 组件层面是基本正确的：你调用了 API 获取分析结果，构建了 alert 对象，并调用了 Store 的方法。

**数据没有写入 `db.json` 的根本原因通常在于 `Pinia Store` (`@/stores/data.ts`) 的实现方式。**

Vue 的状态管理（Pinia）默认只是存储在**内存**中的。如果你在 Store 的 `createAlert` action 中只是把数据 `push` 到了数组里，它只会更新当前页面的显示，而不会发送网络请求去修改后端的 `db.json` 文件。

你需要检查并修改以下三个地方：

### 1. 修改 Pinia Store (`src/stores/data.ts`)

这是最可能遗漏的地方。你需要确保 `createAlert` 方法中显式调用了 `axios.post` 或 `fetch` 发送请求给 `json-server`。

```typescript
// src/stores/data.ts
import { defineStore } from 'pinia'
import axios from 'axios' // 确保安装了 axios

export const useDataStore = defineStore('data', {
  state: () => ({
    alerts: [],
    // ... 其他状态
  }),
  actions: {
    // 核心：这个方法必须发送网络请求
    async createAlert(newAlertPayload: any) {
      try {
        // 1. 补全数据：json-server 不会自动生成 time，通常由前端生成
        const alertData = {
          ...newAlertPayload,
          time: Date.now(), // 必须补充时间戳
          // id: json-server 会自动生成 id，不需要前端传
        }

        // 2. 发送 POST 请求到 json-server (假设端口是 3000)
        // 注意：URL 必须对应你 db.json 中的 key ("alerts")
        const response = await axios.post('http://localhost:3000/alerts', alertData)

        // 3. 请求成功后，再更新本地状态（将后端返回的带 id 的完整数据推入数组）
        this.alerts.push(response.data)
        
        return response.data
      } catch (error) {
        console.error('保存预警失败:', error)
        throw error // 抛出错误以便组件捕获
      }
    }
  }
})
```

### 2. 检查 API 模拟返回结构 (`src/api/analysis.ts`)

在你的组件代码中：

```typescript
const aiResult = response.data.result 
const aiConfidence = response.data.confidence 
```

如果你的 `analyzeImage` 返回的数据结构不包含 `data.result`，代码会在这一行报错并停止执行，导致后面的 `store.createAlert` 永远不会被调用。

确保你的 API 模拟代码如下：

```typescript
// src/api/analysis.ts
export const analyzeImage = (data: any) => {
  return new Promise((resolve) => {
    setTimeout(() => {
      // 模拟后端返回的数据结构，必须包含 data 对象
      resolve({
        data: {
          result: '桃疮痂病', // 模拟结果
          confidence: 0.98    // 模拟置信度
        }
      })
    }, 1500)
  })
}
```

### 4. 乐观更新

手动把返回的 `newAlert` 推入本地数组（这叫“乐观更新”或“手动同步”），这样反应更快：

```typescript
// src/stores/data.ts 优化建议

async function createAlert(alertData: Partial<Alert>) {
    // ... 前面的代码不变 ...
    const res = await http.post('/alerts', payload)
    
    // 优化：直接将后端返回的新对象（包含 id）添加到本地数组，不需要重新请求整个列表
    const newAlert = res.data
    alerts.value.unshift(newAlert) // unshift 加到最前面，或者 push 加到最后面

    return newAlert
}
```
