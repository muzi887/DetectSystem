> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 每次智能分析都落一条可查询的识别记录，支持统计、最近列表、用户纠错回流到 `hard_cases`，并把 `pointId` / 模型版本接到首页与关于页。

**Architecture:** `/api/analysis/*` 已由 Vite 代理到 Flask:5000，历史与统计也放在 Flask，避免再拆 Mock 代理。记录写入 `ml-bjj/serving/data/analysis_records.json`（JSON 列表，进程内读写加锁）。`analyze_image` 成功后自动 `append`；纠错接口把原图拷到 `ml-bjj/data/hard_cases/pending/<正确类名>/`。Mock:3000 仍只管预警 CRUD。

**Tech Stack:** Flask、pytest、Vue 3、现有 `http` 客户端。不新增数据库。

**Spec:** [`新模型训后-后端丰富说明.md`](../方案/新模型训后-后端丰富说明.md) §4.3、§5.2 地块关联、§六。前置：P0/P1 已合并（topk / treatments / 作物过滤）。

- 路由前缀保持 `/api/analysis`，全部走 Flask，不改 Vite `/api/analysis` 代理。
- 不上 Django/Spring；不把训练脚本搬进 serving。
- 本计划不做 batch、ONNX、gunicorn/waitress、病名×气象规则链（P3）。
- 不改智能分析页三列布局；前端只加 `pointId`、纠错、首页最近识别、关于页模型版本。

- `hard_cases` 目录：`ml-bjj/data/hard_cases/pending/<中文类名>/`；类名必须 ∈ `CANONICAL_CLASSES`。

```python
from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path
from threading import Lock

_LOCK = Lock()


def _read(path: Path) -> list[dict]:
    if not path.is_file():
        return []
    raw = path.read_text(encoding="utf-8").strip()
    if not raw:
        return []
    data = json.loads(raw)
    return data if isinstance(data, list) else []


def _write(path: Path, rows: list[dict]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(rows, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def list_records(path: Path) -> list[dict]:
    with _LOCK:
        return _read(path)


def append_record(path: Path, record: dict) -> dict:
    with _LOCK:
        rows = _read(path)
        next_id = (rows[-1]["id"] + 1) if rows else 1
        row = {
            **record,
            "id": next_id,
            "createdAt": record.get("createdAt")
            or datetime.now(timezone.utc).isoformat(),
            "correctedLabel": record.get("correctedLabel"),
        }
        rows.append(row)
        _write(path, rows)
        return row


def recent_records(path: Path, limit: int = 20) -> list[dict]:
    rows = list_records(path)
    ordered = sorted(rows, key=lambda item: item.get("createdAt") or "", reverse=True)
    return ordered[: max(0, limit)]


def stats_by_label(path: Path) -> dict:
    rows = list_records(path)
    tally: dict[str, int] = {}
    for row in rows:
        label = str(row.get("label") or "未知")
        tally[label] = tally.get(label, 0) + 1
    counts = [
        {"label": label, "count": count}
        for label, count in sorted(tally.items(), key=lambda item: (-item[1], item[0]))
    ]
    return {"total": len(rows), "counts": counts}
```

```python
def update_record(path: Path, record_id: int, **fields) -> dict | None:
    with _LOCK:
        rows = _read(path)
        found = None
        for row in rows:
            if row.get("id") == record_id:
                row.update(fields)
                found = row
        _write(path, rows)
        return found
```

```python
@app.route("/api/analysis/feedback", methods=["POST"])
def analysis_feedback():
    label = (request.form.get("correctedLabel") or "").strip()
    if label not in CANONICAL_CLASSES:
        return jsonify({"error": "correctedLabel 不在 23 类中"}), 400
    if "file" not in request.files:
        return jsonify({"error": "未找到文件"}), 400
    file = request.files["file"]
    validate_upload(file)
    dest_dir = hard_cases_pending_root() / label
    dest_dir.mkdir(parents=True, exist_ok=True)
    stamp = datetime.utcnow().strftime("%Y%m%dT%H%M%S")
    suffix = Path(file.filename or "img.jpg").suffix.lower() or ".jpg"
    dest = dest_dir / f"{stamp}_{Path(file.filename or 'img').stem}{suffix}"
    file.save(dest)
    record_id = request.form.get("recordId")
    parsed_id = int(record_id) if record_id and record_id.isdigit() else None
    if parsed_id is not None:
        update_record(records_path(), parsed_id, correctedLabel=label)
    return jsonify({"ok": True, "savedPath": str(dest), "recordId": parsed_id}), 200
```
