MD_LINK = re.compile(r"\[([^\]]*)\]\(([^)]+)\)")
REPO_DOC_PATH = re.compile(r"docs/互联网\+/[^\s)\]`'\"<>]+")


def posix(p: Path) -> str:
    return p.as_posix()


def rel_link(from_dir: Path, to_file: Path) -> str:
    r = Path(os.path.relpath(to_file, from_dir)).as_posix()
    if not r.startswith("."):
        r = "./" + r
    return r


def resolve_existing(path: Path) -> Path:
    return path.resolve()


def main() -> None:
    old_to_new: dict[Path, Path] = {}
    new_to_old: dict[Path, Path] = {}
    by_name: dict[str, list[Path]] = defaultdict(list)

    for old_rel, new_rel in MOVES.items():
        old_p = resolve_existing(BASE / old_rel)
        new_p = (BASE / new_rel).resolve()
        if not old_p.exists():
            raise SystemExit(f"missing source: {old_rel}")
        old_to_new[old_p] = new_p
        new_to_old[new_p] = old_p
        by_name[old_p.name].append(new_p)

    # identity for unmoved files under BASE
    for p in BASE.rglob("*"):
        if p.is_file():
            rp = resolve_existing(p)
            if rp not in old_to_new:

                by_name[rp.name].append(rp)

        return old_to_new.get(old_abs, old_abs)

    def lookup_target(source_old: Path, link_path: str) -> Path | None:
        path_part = link_path.split("#", 1)[0]
        if not path_part:
            return None
        if path_part.startswith(("http://", "https://", "mailto:")):
            return None

        candidate = (source_old.parent / path_part)
        try:
            cand_res = candidate.resolve()
        except OSError:
            cand_res = candidate

        mapped = old_to_new.get(cand_res)
        if mapped is not None:
            return mapped
        if cand_res.exists():
            return cand_res

        name = Path(path_part).name
        hits = by_name.get(name, [])
        unique_new = list(dict.fromkeys(hits))
        if len(unique_new) == 1:
            return unique_new[0]

        norm = path_part.replace("\\", "/")
        if "小挑/" in norm:
            idx = norm.find("小挑/")
            p = (ROOT / "docs" / norm[idx:]).resolve()
            if p.exists():
                return p
        if norm.startswith("docs/"):
            p = (ROOT / norm).resolve()
            if p.exists() or p in old_to_new:
                return old_to_new.get(p, p)
            # try filename under BASE
            if name in by_name and len(set(by_name[name])) == 1:
                return by_name[name][0]
        return None

    def rewrite_md(path: Path, source_old: Path) -> bool:
        text = path.read_text(encoding="utf-8")
        original = text

        def repl_md(m: re.Match[str]) -> str:
            label, url = m.group(1), m.group(2)
            if url.startswith(("http://", "https://", "mailto:")):
                return m.group(0)
            if url.startswith("#"):
                return m.group(0)
            path_part, hash_sep, anchor = url.partition("#")
            if not path_part:
                return m.group(0)
            target = lookup_target(source_old, path_part)
            if target is None:
                return m.group(0)
            new_url = rel_link(path.parent, target)
            if hash_sep:
                new_url += "#" + anchor
            return f"[{label}]({new_url})"

        text = MD_LINK.sub(repl_md, text)

            # docs/互联网+/...

                        mapped = BASE / MOVES[suffix]

                        return m.group(0)

        if text != original:
            path.write_text(text, encoding="utf-8", newline="\n")
            return True
        return False

    extra_files = [
        ROOT / "docs" / "小挑" / "Intro" / "README.md",
        ROOT / "ml-bjj" / "scripts" / "prepare_from_wheat_plantvillage.py",
    ]

        old = new_to_old.get(p.resolve())
        if old is None:
            # unmoved, or extra file
            old = p.resolve()
        if rewrite_md(p, old):
            changed += 1
            print(f"updated links: {p.relative_to(ROOT)}")

        "docs/互联网+/模型/方案/京津冀AI模型精简方案-v3.md",
    )

        py.write_text(py_new, encoding="utf-8", newline="\n")
        print("updated links: ml-bjj/scripts/prepare_from_wheat_plantvillage.py")

    print(f"done. md files with link updates: {changed}")


if __name__ == "__main__":
    main()

这是一份**一次性文档搬家脚本**：把 `docs/互联网+` 下的文件按「赛事 / 规划 / 模型 / 网站」等分类挪到子目录，然后把 Markdown 里的相对链接和仓库路径字符串改成新位置。

# 文档搬家脚本的解释

整体分四块：路径常量与搬家表 → 两个正则 → 辅助函数 → `main()`（搬家 + 改链接）。

### 64–65：两个正则

```64:65:docs/_reorg_internet_plus.py
MD_LINK = re.compile(r"\[([^\]]*)\]\(([^)]+)\)")
REPO_DOC_PATH = re.compile(r"docs/互联网\+/[^\s)\]`'\"<>]+")
```

- **64**：匹配 Markdown 链接 `[文字](url)`。
  - `\[([^\]]*)\]`：方括号里的标签，不能含 `]`。
  - `\(([^)]+)\)`：圆括号里的 URL，不能含 `)`。
  - 两个捕获组后面分别叫 `label` 和 `url`。
- **65**：匹配正文里的仓库路径 `docs/互联网+/...`（不一定包在 Markdown 链接里）。
  - `互联网\+`：`+` 在正则里是量词，必须转义。
  - `[^\s)\]`'\"<>]+`：路径一直吃到空白、`)`、`]`、反引号、引号、尖括号为止，避免把后面的标点吞进去。

- **91–92**：源文件不存在就立刻退出，避免搬一半。
- **93**：记下 old → new。
- **94**：记下 new → old。
- **95**：按**旧文件名**登记新路径。文件搬家一般不改名，所以 `old_p.name == new_p.name`。
